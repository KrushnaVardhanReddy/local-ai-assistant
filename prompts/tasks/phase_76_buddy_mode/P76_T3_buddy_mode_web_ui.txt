        <div class="w-full md:w-80 glass border-l-0 md:border-l border-t md:border-t-0 border-white/5 flex flex-col shrink-0 absolute md:relative bottom-0 left-0 h-40 md:h-full z-20">

                <h2 class="text-sm font-semibold text-primary mb-1">Send a Hint</h2>
                <p class="text-xs text-gray-400">Hints appear privately on your friend's screen.</p>
            </div>
            <div class="p-4 flex-1 flex flex-col gap-3">
                <textarea id="hint-input" class="w-full flex-1 bg-black/40 border border-white/10 rounded-lg p-3 text-sm focus:outline-none focus:border-primary/50 resize-none transition-colors" placeholder="Type your hint here... (e.g. Mention Redis for caching)"></textarea>

    <script>
        const urlParams = new URLSearchParams(window.location.search);
        const token = urlParams.get('token');
        
        const statusText = document.getElementById('status-text');
        const transcriptContainer = document.getElementById('transcript-container');
        const hintInput = document.getElementById('hint-input');
        const sendBtn = document.getElementById('send-btn');

        if (!token) {
            statusText.textContent = "Error: Invalid link (Missing token).";
            statusText.classList.add("text-red-400");
        } else {
            connectWebSocket();
        }

        let ws;
        function connectWebSocket() {
            const wsProtocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
            ws = new WebSocket(`${wsProtocol}//${window.location.host}/ws?token=${token}`);

            ws.onopen = () => {
                statusText.textContent = "🟢 Live - Securely connected";
                statusText.classList.remove("text-gray-400", "text-red-400");
                statusText.classList.add("text-primary");
                hintInput.disabled = false;
            };

            ws.onmessage = (event) => {
                try {
                    const msg = JSON.parse(event.data);
                    if (msg.type === 'transcript') {
                        appendTranscript(msg.speaker, msg.text);
                    }
                } catch (e) { console.error(e); }
            };

            ws.onclose = () => {
                statusText.textContent = "🔴 Disconnected";
                statusText.classList.remove("text-primary");
                statusText.classList.add("text-red-400");
                hintInput.disabled = true;
                sendBtn.disabled = true;
                sendBtn.classList.add("opacity-50", "cursor-not-allowed");
                setTimeout(connectWebSocket, 3000); // Auto-reconnect
            };
        }

        function appendTranscript(speaker, text) {
            // Remove the "Waiting" placeholder if present
            const placeholder = transcriptContainer.querySelector('.text-center.text-gray-500');
            if (placeholder) placeholder.remove();

            const div = document.createElement('div');
            div.className = "mb-4 animate-[fadeIn_0.3s_ease]";
            
            const isInterviewer = speaker === 'interviewer';
            const nameColor = isInterviewer ? 'text-blue-400' : 'text-purple-400';
            const nameLabel = isInterviewer ? 'Interviewer' : 'Candidate (Friend)';
            
            div.innerHTML = `
                <div class="text-[0.65rem] font-bold uppercase tracking-wider ${nameColor} mb-1">${nameLabel}</div>
                <div class="text-sm bg-white/5 p-3 rounded-lg border border-white/5 inline-block max-w-[90%] leading-relaxed">${text}</div>
            `;
            
            transcriptContainer.appendChild(div);
            transcriptContainer.scrollTop = transcriptContainer.scrollHeight;
        }

        hintInput.addEventListener('input', () => {
            if (hintInput.value.trim().length > 0 && ws && ws.readyState === WebSocket.OPEN) {
                sendBtn.disabled = false;
                sendBtn.classList.remove("opacity-50", "cursor-not-allowed");
            } else {
                sendBtn.disabled = true;
                sendBtn.classList.add("opacity-50", "cursor-not-allowed");
            }
        });

        function sendHint() {
            const text = hintInput.value.trim();
            if (!text || !ws || ws.readyState !== WebSocket.OPEN) return;
            
            ws.send(JSON.stringify({ type: 'hint', text: text }));
            
            // Visual feedback
            const originalText = sendBtn.textContent;
            sendBtn.textContent = "Sent!";
            sendBtn.classList.add("bg-green-500");
            setTimeout(() => {
                sendBtn.textContent = originalText;
                sendBtn.classList.remove("bg-green-500");
            }, 1000);
            
            hintInput.value = '';
            sendBtn.disabled = true;
            sendBtn.classList.add("opacity-50", "cursor-not-allowed");
        }

        sendBtn.addEventListener('click', sendHint);
        hintInput.addEventListener('keypress', (e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                sendHint();
            }
        });
    </script>
</body>
</html>
```

─── STEP 2: Serve the HTML in `backend/buddy/server.go` ───────

In `wails-app/backend/buddy/server.go` (which is being created in P76-T1), you need to embed the HTML file.

2. Add the `//go:embed` directive and variable:

    "bufio"
    "context"
    "crypto/rand"
    "embed" // NEW
    // ...
)

//go:embed index.html
var staticAssets embed.FS
```

3. Update the `Start()` method's `http.ServeMux` to serve the embedded file on `/`:
```go
    // Set up HTTP routes
    mux := http.NewServeMux()
    mux.HandleFunc("/ws", s.handleWebSocket)
    mux.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
        w.WriteHeader(http.StatusOK)
        w.Write([]byte("ok"))
    })
    
    // Serve the embedded index.html on root
    mux.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
        if r.URL.Path != "/" {
            http.NotFound(w, r)
            return
        }
        content, err := staticAssets.ReadFile("index.html")
        if err != nil {
            http.Error(w, "File not found", http.StatusInternalServerError)
            return
        }
        w.Header().Set("Content-Type", "text/html; charset=utf-8")
        w.Write(content)
    })
```
