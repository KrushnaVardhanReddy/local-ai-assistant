You are tasked with implementing full remote control of BarnOwl AI via the Buddy Mode WebSocket. This is Phase 79, Task 1 (P79-T1).

════════════════════════════════════════════════════════════════
TASK CONTEXT
════════════════════════════════════════════════════════════════

In Buddy Mode, a friend currently receives the live transcript and can send text hints.
This task upgrades Buddy Mode to a **full remote control system**. The friend should be able to
pilot the entire app from their browser without the candidate ever needing to touch the keyboard or
mouse. This is critical for stealth — the candidate just listens while their friend drives.

The architecture uses the existing WebSocket channel on port `:8765` (created in P76-T1).
The friend's HTML page sends a `remote_action` message, the Go buddy server receives it,
and routes it to a new `onRemoteAction` callback in `app.go`, which dispatches to the
appropriate existing Wails-bound method.

```go
// handleBuddyRemoteAction dispatches incoming buddy remote control actions
// to the corresponding app methods. Runs in a goroutine.
func (a *App) handleBuddyRemoteAction(action, payload string) {
    log.Printf("[Buddy] Remote action received: %s (payload: %q)", action, payload)
    switch action {
    case "flush_llm":
        if err := a.FlushQuestionBuffer(); err != nil {
            log.Printf("[Buddy] flush_llm error: %v", err)
        }
    case "clear_state":
        a.ClearState()
    case "toggle_mic":
        isOn := a.ToggleMic()
        // Emit state back to buddy clients so their button updates
        a.buddyServer.BroadcastState(map[string]interface{}{"micActive": isOn})
    case "toggle_manual_mode":
        a.cmdMutex.Lock()
        a.isManualMode = !a.isManualMode
        newMode := a.isManualMode
        a.cmdMutex.Unlock()
        a.SetManualMode(newMode)
        a.buddyServer.BroadcastState(map[string]interface{}{"manualMode": newMode})
    case "send_chat":
        if payload != "" {
            if err := a.SendChat(payload); err != nil {
                log.Printf("[Buddy] send_chat error: %v", err)
            }
        }
    case "append_to_buffer":
        if payload != "" {
            if err := a.AppendToBuffer(payload); err != nil {
                log.Printf("[Buddy] append_to_buffer error: %v", err)
            }
        }
    case "capture_screen":
        screenshot := a.CaptureScreen()
        if screenshot != "" {
            if err := a.AnalyzeVision(screenshot, ""); err != nil {
                log.Printf("[Buddy] capture_screen AnalyzeVision error: %v", err)
            }
        }
    case "end_session":
        if _, err := a.EndSession(); err != nil {
            log.Printf("[Buddy] end_session error: %v", err)
        }
    default:
        log.Printf("[Buddy] Unknown remote action: %s", action)
    }
}
```

Note: You need to check if `isManualMode` field exists on the `App` struct. If it does not already
exist as a field on `App`, add it: `isManualMode bool`. If there is already a way to query the
current manual mode state from the engine, use that instead.

─── STEP 5: Add `BroadcastState` to `server.go` ────────────────

```go
// BroadcastState sends a state update message to all connected buddy clients.
func (s *Server) BroadcastState(state map[string]interface{}) {
    data := map[string]interface{}{
        "type":  MsgTypeState,
        "state": state,
    }
    jsonData, err := json.Marshal(data)
    if err != nil {
        return
    }
    s.mu.Lock()
    defer s.mu.Unlock()
    for client := range s.clients {
        _ = client.WriteMessage(websocket.TextMessage, jsonData)
    }
}
```

function sendAction(action, payload = '') {
    if (!ws || ws.readyState !== WebSocket.OPEN) return;
    ws.send(JSON.stringify({ type: 'remote_action', action, payload }));
    
    // Flash button feedback
    const btn = document.getElementById(`btn-${action.replace('_', '-')}`);
    if (btn) {
        const orig = btn.textContent;
        btn.textContent = '✓ Sent';
        btn.disabled = true;
        setTimeout(() => { btn.textContent = orig; btn.disabled = false; }, 1000);
    }
}

function sendChat() {
    const input = document.getElementById('chat-input');
    const text = input?.value?.trim();
    if (!text) return;
    sendAction('send_chat', text);
    input.value = '';
}

// Handle state updates from server
ws.onmessage = (event) => {
    const msg = JSON.parse(event.data);
    if (msg.type === 'transcript') {
        appendTranscript(msg.speaker, msg.text);
    } else if (msg.type === 'state') {
        // Update button labels based on remote state
        if (msg.state?.micActive !== undefined) {
            const btn = document.getElementById('btn-mic');
            if (btn) btn.textContent = msg.state.micActive ? '🔴 Stop Mic' : '🎙️ Toggle Mic';
        }
        if (msg.state?.manualMode !== undefined) {
            const btn = document.getElementById('btn-manual');
            if (btn) btn.textContent = msg.state.manualMode ? '🤖 Enable Auto' : '⏸️ Go Manual';
        }
    }
};
