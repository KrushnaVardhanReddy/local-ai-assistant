STEP-BY-STEP IMPLEMENTATION INSTRUCTIONS
═══════════════════════════════════════════════════════════════

1. In `IDEShell.svelte`:
   - Synchronize `isExplorerOpen`:
     - Allow `isExplorerOpen` to toggle when `activeAction === 'explorer'`.
     - Ensure `handleSidebarToggle` and `handleAction` keep the drawer state responsive.

2. In `InterviewHUD.svelte`:
   - In `handleAction(action)`:
     ```ts
     function handleAction(action: string) {
       if (action === 'explorer') {
         activeAction = activeAction === 'explorer' ? '' : 'explorer';
       } else if (action === 'ears') {
         activeAction = 'ears';
         activeDrawer = 'ears';
       } else if (action === 'copilot') {
         activeAction = 'copilot';
         activeDrawer = 'copilot';
       } else if (action === 'mock') {
         handleMockModeToggle();
       } else if (action === 'keys') {
         uiState.hotkeysPanelOpen = !uiState.hotkeysPanelOpen;
       } else if (action === 'settings') {
         activeAction = activeAction === 'settings' ? activeDrawer : 'settings';
       } else {
         activeAction = action;
       }
     }
     ```
   - Add `refreshIDEState()`:
     ```ts
     async function refreshIDEState() {
       if (App?.GetIDEState) {
         const state = await App.GetIDEState();
         if (state) {
           includeActiveDocContext = state.includeActiveDocContext;
           activeDocumentName = state.activeDocumentName || '';
           recordExpanded(workspaceTree);
           workspaceTree = mapTree(state.workspaceTree);
           openTabs = state.openDocuments || [];
           activeDocumentPath = state.activeDocumentPath || '';
           activeDocumentContent = state.activeDocumentContent || '';
         }
       }
     }
     ```
   - Call `await refreshIDEState()` after `PromptOpenFile()`, `PromptOpenDirectory()`,
     `OpenFile()`, `SetActiveDocument()`, and `CloseDocument()`.
