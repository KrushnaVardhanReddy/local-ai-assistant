═══════════════════════════════════════════════════════════════
OBJECTIVE
═══════════════════════════════════════════════════════════════
With the StealthEngine in place (P54-T5), the first new product can be
created as a thin "skin" — a different system prompt, a different Wails
App shell, and a new Svelte UI. No engine code is changed.

StealthPresenter is an invisible teleprompter HUD that:
  - Sits near the user's webcam, excluded from screen capture
  - Loads presentation materials (PDF, PPTX, DOCX, TXT, MD) as script/context
  - Auto-scrolls based on voice (using the existing Whisper STT)
  - Optionally uses the LLM to answer live audience questions
  - Is completely invisible in Zoom, Teams, Google Meet, OBS

This task creates the product shell and a minimal Svelte UI scaffold.
It does NOT implement full PDF rendering or voice-scroll (those are
future tasks). It validates the full product wiring end-to-end.

Prerequisite: P54-T5 must be merged.

2. CREATE: wails-app/products/presenter/app.go
   Package: `presenter`

   import (
       "wails-app/adapters/cache"
       "wails-app/adapters/events"
       "wails-app/adapters/llm"
       "wails-app/backend/audio"
       "wails-app/backend/stt"
       "wails-app/backend"
       "wails-app/core/engine"
   )

   // PresenterApp is the Wails App struct for the StealthPresenter product.
   // It is thin — it owns only Wails-facing concerns (context, window, audio).
   // All pipeline logic is delegated to the embedded StealthEngine.
   type PresenterApp struct {
       ctx          context.Context
       engine       *engine.StealthEngine
       audioCapture driven.AudioCapturePort
   }

   func NewPresenterApp() *PresenterApp {
       // Build adapters
       db, _ := backend.NewVectorDB("./data/presenter_cache.db")
       llmAdapter   := llmadapter.NewOpenAIAdapter()
       cacheAdapter := cacheadapter.NewSQLiteVecAdapter(db)

       // STT: prefer Groq if key set, else fall back to local Whisper
       var sttEngine stt.STTEngine
       if key := os.Getenv("GROQ_API_KEY"); key != "" {
           sttEngine = stt.NewGroqEngine(key, "whisper-large-v3-turbo")
       } else {
           validPath, err := stt.EnsureWhisperModel("")
           if err == nil {
               sttEngine, _ = stt.LoadWhisperEngine(validPath)
           }
       }

       eng := engine.New(
           engine.Config{SystemPrompt: SystemPrompt},
           stt.NewSTTManager(sttEngine),
           llmAdapter,
           cacheAdapter,
           eventsadapter.NewNoopEventAdapter(), // replaced in startup()
       )

       return &PresenterApp{
           engine:       eng,
           audioCapture: audio.NewCaptureEngine(),
       }
   }

   // Optional: Provide a constructor for testing that accepts the port
   func NewPresenterAppWithPorts(eng *engine.StealthEngine, ac driven.AudioCapturePort) *PresenterApp {
       return &PresenterApp{engine: eng, audioCapture: ac}
   }

   // startup is called by Wails on app start.
   func (p *PresenterApp) startup(ctx context.Context) {
       p.ctx = ctx
       p.engine.SetEventsAdapter(eventsadapter.NewWailsEventAdapter(ctx))
       // Hide from taskbar for stealth
       _ = window.HideFromTaskbar(ctx)
       // Auto-start microphone
       _ = p.audioCapture.Initialize()
       go p.audioCapture.StartCapture(-1, false, func(samples []float32) {
           _ = p.engine.ProcessAudio(samples)
       })
   }

   // Wails-exposed methods (called from Svelte frontend):
   func (p *PresenterApp) GetState() map[string]interface{} { ... } // delegate to engine
   func (p *PresenterApp) ClearState()                      { p.engine.ClearState() }
   func (p *PresenterApp) AskQuestion(q string) error       { return p.engine.AskQuestion(q) }
